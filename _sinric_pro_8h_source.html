240void SinricProClass::begin(String appKey, String appSecret, String serverURL) {
241 bool success = true;
242
243 // Validate APP_KEY
244 // - Must be 36 characters
245 // - UUID format: xxxxxxxx-xxxx-xxxx-xxxx-xxxxxxxxxxxx
246 if (appKey.length() != 36) {
247 Serial.printf("[SinricPro:begin()]: Invalid App Key '%s' detected (Length: %d; Expected: 36). Initialization aborted! Please check your SinricPro credentials.\r\n", appKey.c_str(), appKey.length());
248 success = false;
249 } else if (appKey.charAt(8) != '-' || appKey.charAt(13) != '-' || appKey.charAt(18) != '-' || appKey.charAt(23) != '-') {
250 Serial.printf("[SinricPro:begin()]: App Key '%s' is in an invalid format. Initialization aborted!\r\n", appKey.c_str());
251 success = false;
252 }
253
254 // Validate APP_SECRET (73 characters)
255 // - Must be 73 characters
256 // - Double UUID format: xxxxxxxx-xxxx-xxxx-xxxx-xxxxxxxxxxxx-xxxxxxxx-xxxx-xxxx-xxxx-xxxxxxxxxxxx
257 if (appSecret.length() != 73) {
258 Serial.printf("[SinricPro:begin()]: Invalid App Secret '%s' detected (Length: %d; Expected: 73). Initialization aborted! Please check your SinricPro credentials.\r\n", appSecret.c_str(), appSecret.length());
259 success = false;
260 } else if (appSecret.charAt(8) != '-' || appSecret.charAt(13) != '-' || appSecret.charAt(18) != '-' || appSecret.charAt(23) != '-' ||
261 appSecret.charAt(36) != '-' || appSecret.charAt(45) != '-' || appSecret.charAt(50) != '-' || appSecret.charAt(55) != '-' || appSecret.charAt(60) != '-') {
262 Serial.printf("[SinricPro:begin()]: App Secret '%s' is in an invalid format. Initialization aborted!\r\n", appSecret.c_str());
263 success = false;
264 }
265
266 if (!success) {
267 _begin = false;
268 return;
269 }
270
271 this->appKey = appKey;
272 this->appSecret = appSecret;
273 this->serverURL = serverURL;
274 _begin = true;
275#ifndef SINRICPRO_NO_LOCAL_CONTROL
276 _udpListener.begin(&receiveQueue);
277#endif
278#ifdef SINRICPRO_MDNS_ENABLED
279 {
280 String hostName = "sinricpro-" + WiFi.macAddress();
281 hostName.replace(":", "");
282 hostName.toLowerCase();
283 _mdnsListener.begin(hostName, joinDeviceIds(',')); //mDNS TXT uses ',' (CSV).
284 _mdnsDeviceCount = devices.size();
285 }
286#endif
287}

289template <typename DeviceType>
290DeviceType& SinricProClass::add(String deviceId) {
291 // Validate DEVICE_ID
292 // - Must be 24 characters
293 // - Hexadecimal only (0-9, a-f, A-F)
294 // - Format: 695b4624f6e5944047661b8a
295 if (deviceId.length() != 24) {
296 Serial.printf("[SinricPro:add()]: Device Id \"%s\" is invalid (wrong length: expected 24, got %d)!! Please check your device-id!!\r\n", deviceId.c_str(), deviceId.length());
297 }
298
299 DeviceType* newDevice = new DeviceType(deviceId);
300 DEBUG_SINRIC("[SinricPro:add()]: Adding device with id \"%s\".\r\n", deviceId.c_str());
301 newDevice->begin(this);
302
303 devices.push_back(newDevice);
304 return *newDevice;
305}

332void SinricProClass::handle() {
333 static bool begin_error = false;
334 if (!_begin) {
335 if (!begin_error) { // print this only once!
336 DEBUG_SINRIC("[SinricPro:handle()]: ERROR! SinricPro.begin() failed or was not called prior to event handler\r\n");
337 DEBUG_SINRIC("[SinricPro:handle()]: -Reasons include an invalid app-key, invalid app-secret or no valid deviceIds)\r\n");
338 DEBUG_SINRIC("[SinricPro:handle()]: -SinricPro is disabled! Check earlier log messages for details.\r\n");
339 begin_error = true;

342 }

344 if (!isConnected()) connect();
345 _websocketListener.handle();

351#endif
352
353 handleReceiveQueue();
354 handleSendQueue();

357JsonDocument SinricProClass::prepareRequest(String deviceId, const char* action) {
358 JsonDocument requestMessage;
359 JsonObject header = requestMessage[FSTR_SINRICPRO_header].to<JsonObject>();
360 header[FSTR_SINRICPRO_payloadVersion] = 2;
361 header[FSTR_SINRICPRO_signatureVersion] = 1;
362
363 JsonObject payload = requestMessage[FSTR_SINRICPRO_payload].to<JsonObject>();
364 payload[FSTR_SINRICPRO_action] = action;
365 payload[FSTR_SINRICPRO_createdAt] = 0;
366 payload[FSTR_SINRICPRO_deviceId] = deviceId;
367 payload[FSTR_SINRICPRO_replyToken] = MessageID().getID();
368 payload[FSTR_SINRICPRO_type] = FSTR_SINRICPRO_request;
369 payload[FSTR_SINRICPRO_value].to<JsonObject>();
370 return requestMessage;
371}

383void SinricProClass::handleModuleRequest(JsonDocument& requestMessage, const SinricProMessage* origin) {
384 DEBUG_SINRIC("[SinricPro.handleModuleScopeRequest()]: handling module scope request\r\n");
385#ifndef NODEBUG_SINRIC
386 serializeJsonPretty(requestMessage, DEBUG_ESP_PORT);
387#endif
388
389 JsonDocument responseMessage = prepareResponse(requestMessage);
390
391 String action = requestMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_action] | "";
392 JsonObject request_value = requestMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_value];
393 JsonObject response_value = responseMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_value];
394 SinricProRequest request{action, "", request_value, response_value};
395
396 bool success = _moduleCommandHandler.handleRequest(request);
397
398 responseMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_success] = success;
399 responseMessage[FSTR_SINRICPRO_payload].remove(FSTR_SINRICPRO_deviceId);
400 if (!success) {
401 if (responseMessageStr.length() > 0) {
402 responseMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_message] = responseMessageStr;
403 responseMessageStr = "";
404 } else {
405 responseMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_message] = "Module did not handle \"" + action + "\"";
406 }
407 }
408
409 String responseString;
410 serializeJson(responseMessage, responseString);
411 sendQueue.push(new SinricProMessage(origin->getInterface(), responseString.c_str(),
412 origin->getRemoteIP(), origin->getRemotePort()));
413}

415void SinricProClass::handleDeviceRequest(JsonDocument& requestMessage, const SinricProMessage* origin) {
416 DEBUG_SINRIC("[SinricPro.handleDeviceRequest()]: handling device sope request\r\n");
417#ifndef NODEBUG_SINRIC
418 serializeJsonPretty(requestMessage, DEBUG_ESP_PORT);
419#endif
420
421 JsonDocument responseMessage = prepareResponse(requestMessage);
422
423 // handle devices
424 bool success = false;
425 const char* deviceId = requestMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_deviceId];
426 String action = requestMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_action] | "";
427 String instance = requestMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_instanceId] | "";
428 JsonObject request_value = requestMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_value];
429 JsonObject response_value = responseMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_value];
430
431 bool deviceIsOurs = false;
432
433 for (auto& device : devices) {
434 if (device->getDeviceId() == deviceId && success == false) {
435 deviceIsOurs = true;
436 SinricProRequest request{
437 action,
438 instance,
439 request_value,
440 response_value};
441 success = device->handleRequest(request);
442 responseMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_success] = success;
443 if (!success) {
444 if (responseMessageStr.length() > 0) {
445 responseMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_message] = responseMessageStr;
446 responseMessageStr = "";
447 } else {
448 responseMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_message] = "Device did not handle \"" + action + "\"";
449 }
450 }
451 }
452 }
453
454 if (!deviceIsOurs && origin->getInterface() == IF_UDP) {
455 DEBUG_SINRIC("[SinricPro.handleDeviceRequest()]: ignoring LAN request for unknown device \"%s\"\r\n", deviceId);
456 return;
457 }
458
459 String responseString;
460 serializeJson(responseMessage, responseString);
461 sendQueue.push(new SinricProMessage(origin->getInterface(), responseString.c_str(),
462 origin->getRemoteIP(), origin->getRemotePort()));
463}

465void SinricProClass::handleReceiveQueue() {
466 if (receiveQueue.size() == 0) return;
467
468 DEBUG_SINRIC("[SinricPro.handleReceiveQueue()]: %i message(s) in receiveQueue\r\n", receiveQueue.size());
469 while (receiveQueue.size() > 0) {
470 SinricProMessage* rawMessage = receiveQueue.front();
471 receiveQueue.pop();
472 JsonDocument jsonMessage;
473 deserializeJson(jsonMessage, rawMessage->getMessage());
474
475 bool sigMatch = false;
476
477 if (strncmp(rawMessage->getMessage(), "{\"timestamp\":", 13) == 0 && strlen(rawMessage->getMessage()) <= 26) {
478 sigMatch = true; // timestamp message has no signature...ignore sigMatch for this!
479 } else {
480 String signature = jsonMessage[FSTR_SINRICPRO_signature][FSTR_SINRICPRO_HMAC] | "";
481 String payload = extractPayload(rawMessage->getMessage());
482 String calculatedSignature = calculateSignature(appSecret.c_str(), payload);
483 sigMatch = (calculatedSignature == signature);
484 }
485
486 String messageType = jsonMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_type];
487
488 if (sigMatch) { // signature is valid process message
489 DEBUG_SINRIC("[SinricPro.handleReceiveQueue()]: Signature is valid. Processing message...\r\n");
490 extractTimestamp(jsonMessage);
491 if (messageType == FSTR_SINRICPRO_response) handleResponse(jsonMessage);
492 if (messageType == FSTR_SINRICPRO_request) {
493 String scope = jsonMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_scope] | FSTR_SINRICPRO_device;
494 if (strcmp(FSTR_SINRICPRO_module, scope.c_str()) == 0) {
495 handleModuleRequest(jsonMessage, rawMessage);
496 } else {
497 handleDeviceRequest(jsonMessage, rawMessage);
498 }
499 };
500 } else {
501 handleInvalidSignatureRequest(jsonMessage, rawMessage);
502 }
503 delete rawMessage;
504 }
505}

524void SinricProClass::handleSendQueue() {
525 // Visit each pending message once so deferred cloud messages cannot block LAN replies.
526 size_t pending = sendQueue.size();
527 while (pending-- > 0) {
528 DEBUG_SINRIC("[SinricPro:handleSendQueue()]: %i message(s) in sendQueue\r\n", sendQueue.size());
529
530 SinricProMessage* rawMessage = sendQueue.front();
531
532 if (rawMessage->getInterface() == IF_WEBSOCKET &&
533 (!isConnected() || !timestamp.getTimestamp())) {
534 sendQueue.pop();
535 sendQueue.push(rawMessage);
536 continue;
537 }
538
539 DEBUG_SINRIC("[SinricPro:handleSendQueue()]: Sending message...\r\n");
540 sendQueue.pop();
541
542 JsonDocument jsonMessage;
543 deserializeJson(jsonMessage, rawMessage->getMessage());
544 jsonMessage[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_createdAt] = timestamp.getTimestamp();
545 signMessage(appSecret, jsonMessage);
546
547 String messageStr;
548
549 serializeJson(jsonMessage, messageStr);
550#ifndef NODEBUG_SINRIC
551 serializeJsonPretty(jsonMessage, DEBUG_ESP_PORT);
552 Serial.println();
553#endif
554
555 switch (rawMessage->getInterface()) {
556 case IF_WEBSOCKET:
557 if (isConnected()) {
558 DEBUG_SINRIC("[SinricPro:handleSendQueue]: Sending to websocket\r\n");
559 _websocketListener.sendMessage(messageStr);
560 } else {
561 DEBUG_SINRIC("[SinricPro:handleSendQueue]: Dropping WS message — not connected\r\n");
562 }
563 break;
564 case IF_UDP:
565#ifndef SINRICPRO_NO_LOCAL_CONTROL
566 DEBUG_SINRIC("[SinricPro:handleSendQueue]: Sending to UDP\r\n");
567 _udpListener.sendMessage(messageStr, rawMessage->getRemoteIP(), rawMessage->getRemotePort());
568#endif
569 break;
570 default:
571 break;
572 }
573 delete rawMessage;
574 DEBUG_SINRIC("[SinricPro:handleSendQueue()]: message sent.\r\n");
575 }
576}

663bool SinricProClass::sendSettingEvent(String settingId, SettingValue settingValue, String cause) {
664 if (_settingEventLimiter) return false;
665
666 JsonDocument eventMessage = prepareEvent("", FSTR_SETTING_setSetting, cause.c_str());
667 JsonObject payload = eventMessage[FSTR_SINRICPRO_payload];
668
669 payload.remove(FSTR_SINRICPRO_deviceId);
670 payload[FSTR_SINRICPRO_scope] = FSTR_SINRICPRO_module;
671
672 JsonObject event_value = payload[FSTR_SINRICPRO_value];
673 event_value[FSTR_SETTING_id] = settingId;
674
675 if (std::holds_alternative<int>(settingValue)) {
676 event_value[FSTR_SETTING_value] = std::get<int>(settingValue);
677 } else if (std::holds_alternative<float>(settingValue)) {
678 event_value[FSTR_SETTING_value] = std::get<float>(settingValue);
679 } else if (std::holds_alternative<bool>(settingValue)) {
680 event_value[FSTR_SETTING_value] = std::get<bool>(settingValue);
681 } else if (std::holds_alternative<String>(settingValue)) {
682 event_value[FSTR_SETTING_value] = std::get<String>(settingValue);
683 }
684
685 sendMessage(eventMessage);
686 return true;
687}

734void SinricProClass::extractTimestamp(JsonDocument& message) {
735 unsigned long tempTimestamp = 0;
736 // extract timestamp from timestamp message right after websocket connection is established
737 tempTimestamp = message["timestamp"] | 0;
738 if (tempTimestamp) {
739 timestamp.setTimestamp(tempTimestamp);
740 DEBUG_SINRIC("[SinricPro:extractTimestamp(): Got Timestamp %lu\r\n", tempTimestamp);
741 return;
742 }
743
744 // extract timestamp from request message
745 tempTimestamp = message[FSTR_SINRICPRO_payload][FSTR_SINRICPRO_createdAt] | 0;
746 if (tempTimestamp) {
747 DEBUG_SINRIC("[SinricPro:extractTimestamp(): Got Timestamp %lu\r\n", tempTimestamp);
748 timestamp.setTimestamp(tempTimestamp);

750 }

816JsonDocument SinricProClass::prepareResponse(JsonDocument& requestMessage) {
